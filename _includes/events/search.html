{%- assign tags = blank -%} {%- for post in site.posts -%} {%- assign post_tags = post.tags | join:'|' | append:'|' -%} {%- if post_tags != '|' -%} {%- assign tags = tags | append:post_tags -%} {%- endif -%} {%- endfor -%} {%- assign all_tags = tags | split:'|' -%} {%- assign tags = all_tags | uniq | sort -%}
{%- for tag in tags %} {%- assign tag_count = 0 -%} {%- for t in all_tags -%}{%- if t == tag -%}{%- assign tag_count = tag_count | plus: 1 -%}{%- endif -%}{%- endfor %} {%- endfor %}
